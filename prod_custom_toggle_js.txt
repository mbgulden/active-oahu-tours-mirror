
// Nav dropdown + mobile menu toggle
document.addEventListener('DOMContentLoaded', function() {
  // --- Mobile menu toggle (WordPress _s theme pattern) ---
  var navContainer = document.getElementById('site-navigation');
  var toggleBtn = document.querySelector('.menu-toggle');
  if (navContainer && toggleBtn) {
    toggleBtn.addEventListener('click', function(e) {
      e.preventDefault();
      navContainer.classList.toggle('toggled');
      var expanded = toggleBtn.getAttribute('aria-expanded') === 'true' ? 'false' : 'true';
      toggleBtn.setAttribute('aria-expanded', expanded);
    });
  }

  // --- Sub-menu show/hide (desktop only; mobile handled by CSS @media) ---
  var isDesktop = window.innerWidth > 1023;
  var subs = document.querySelectorAll('.sub-menu');
  if (isDesktop) {
    for (var i = 0; i < subs.length; i++) { subs[i].style.display = 'none'; }
  }
  
  function closeSiblings(li, keep) {
    var parent = li.parentNode;
    for (var i = 0; i < parent.children.length; i++) {
      var sib = parent.children[i];
      if (sib !== li) {
        var sibSub = sib.querySelector('.sub-menu');
        if (sibSub && sibSub !== keep) { sibSub.style.display = 'none'; }
      }
    }
  }
  
  var links = document.querySelectorAll('.menu-item-has-children > a');
  for (var i = 0; i < links.length; i++) {
    (function(link) {
      link.addEventListener('click', function(e) {
        if (!isDesktop) return; // Let CSS handle mobile
        var li = link.parentNode;
        var sub = null;
        for (var j = 0; j < li.children.length; j++) {
          if (li.children[j].classList.contains('sub-menu')) { sub = li.children[j]; break; }
        }
        if (sub) {
          e.preventDefault();
          closeSiblings(li, sub);
          sub.style.display = (sub.style.display === 'none') ? 'block' : 'none';
        }
      });
    })(links[i]);
  }
  
  document.addEventListener('click', function(e) {
    if (!isDesktop) return; // Let CSS handle mobile
    var el = e.target;
    var inMenu = false;
    while (el) {
      if (el.classList && el.classList.contains('menu-item-has-children')) { inMenu = true; break; }
      el = el.parentNode;
    }
    if (!inMenu) {
      var all = document.querySelectorAll('.sub-menu');
      for (var i = 0; i < all.length; i++) { all[i].style.display = 'none'; }
    }
  });
});
